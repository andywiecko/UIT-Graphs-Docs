    Plot with boxes

    Data

    To plot boxes, you can use PlotWithBoxes. Unlike other plot types, this plot requires an IReadOnlyList<AABB> data series. To construct a box, i.e. an AABB, you can use the constructor:

    var box = new AABB(min: math.float2(0, 3), max: math.float2(4, 8));

    the static construction method:

    var box = AABB.Construct(math.float2(0, 3), math.float2(4, 8));

    or other methods. See AABB in the API documentation for more details. Below is an example of a plot with randomly generated boxes.

    var graph = new Graph
{
    XAxis = { Min = 0, Max = 50 + 4 },
    YAxis = { Min = 0, Max = 30 + 4 },
    Theme = { Grid = { Display = DisplayStyle.None } },
};

var random = new Unity.Mathematics.Random(seed: 42);
AABB rnd()
{
    var p = random.NextFloat2(min: 0, max: math.float2(50, 30));
    var s = random.NextFloat2(min: 1, max: 8);
    return new(p, p + s);
}
var boxes = Enumerable.Range(0, 30).Select(i => rnd()).ToArray();

graph.PlotWithBoxes(boxes, theme: new() { FillColor = new(1, 0, 0, 0.1f) });


    Styling

    This plot type has the same properties as PlotWithLines, with the addition of the FillColor property (--graph-fill-color).

    --graph-line-dash-pattern: "4 4";
    --graph-fill-color: rgba(1.0, 0, 0, 0.5);

    var graph = new Graph
{
    XAxis = { Min = -2 * math.PI, Max = 2 * math.PI },
    YAxis = { Min = -0.4f, Max = +.4f }
};

static AABB ctor(float x)
{
    var y = 0.1f * math.cos(x);
    return AABB.Construct(
        math.float2(x - 0.5f, +y + 0.15f),
        math.float2(x + 0.5f, -y - 0.15f)
    );
}

var data = SequenceAABB.Uniform(x => ctor(x), samples: 11);
var plot = graph.PlotWithBoxes(data, theme: new()
{
    LineColor = Color.blue,
    LineDashPattern = new float[] { 4, 4 },
    LineWidth = 3,
    FillColor = new Color(1, 0, 0, 0.5f),
});
